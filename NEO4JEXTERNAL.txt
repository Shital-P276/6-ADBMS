   name:'Sam',

CREATE
(s1:Student {name:'Sam'}),
(s2:Student {name:'John'}),
(s3:Student {name:'Alice'});

//CREATE relationship
MATCH
(s1:Student {name:'Sam'}),
(s2:Student {name:'John'})
CREATE
(s1)-[:FRIEND]->(s2);

//TEACHER RELATIONSHIP
MATCH
(s:Student {name:'Sam'}),
(t:Teacher {name:'Prof. Rao'})
CREATE
(s)-[:TAUGHT_BY]->(t);
